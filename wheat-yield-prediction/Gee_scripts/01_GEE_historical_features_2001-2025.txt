      reducer: ee.Reducer.mean(),

// VPD EXTRACTION — March–April Mean
// Source: ERA5-Land Daily Aggregated
// Window: March 1 – April 30 (grain filling + maturity)
// Method: VPD = es(Tair) - ea(Tdew), clamped to >= 0
// Using Tetens formula (FAO standard)
// Units: kPa
// Batch 1: 2001–2010
// =====================================================

var districts = ee.FeatureCollection(
  'projects/ee-ahsanahsan520/assets/multan_districts'
).filter(ee.Filter.inList('NAME_3', [
  'Multan', 'Lodhran', 'Khanewal', 'Vehari'
]));

// Load ERA5-Land air temperature and dew point
var era5 = ee.ImageCollection('ECMWF/ERA5_LAND/DAILY_AGGR')
  .select([
    'temperature_2m',
    'dewpoint_temperature_2m'
  ]);

// =====================================================
// CHANGE ONLY THIS LINE FOR EACH BATCH
// Batch 1: 2001–2010
// Batch 2: 2011–2020
// Batch 3: 2021–2025
// =====================================================
var years = ee.List.sequence(2021, 2025);

var vpdTable = ee.FeatureCollection(
  years.map(function(year) {
    year = ee.Number(year);

    // March 1 to April 30 of same year
    var startDate = ee.Date.fromYMD(year, 3, 1);
    var endDate   = ee.Date.fromYMD(year, 4, 30);

    // Filter to March-April window
    var era5Season = era5
      .filterDate(startDate, endDate.advance(1, 'day'));

    // Step 1: Convert Kelvin to Celsius
    var meanImage = era5Season.mean();

    var tair = meanImage
      .select('temperature_2m')
      .subtract(273.15)
      .rename('Tair_C');

    var tdew = meanImage
      .select('dewpoint_temperature_2m')
      .subtract(273.15)
      .rename('Tdew_C');

    // Step 2: Saturation vapor pressure es(Tair)
    // Tetens formula — FAO standard
    var es = tair
      .expression(
        '0.6108 * exp((17.27 * T) / (T + 237.3))',
        {'T': tair}
      )
      .rename('es_kPa');

    // Step 3: Actual vapor pressure ea(Tdew)
    var ea = tdew
      .expression(
        '0.6108 * exp((17.27 * Td) / (Td + 237.3))',
        {'Td': tdew}
      )
      .rename('ea_kPa');

    // Step 4: VPD = es - ea
    // Clamped to >= 0 (physical constraint)
    // Negative values can occur due to ERA5 rounding errors
    // where Tdew slightly exceeds Tair — not physically real
    var vpd = es
      .subtract(ea)
      .max(0)
      .rename('VPD_kPa');

    // Zonal statistics per district
    var stats = vpd.reduceRegions({
      collection: districts,
      reducer: ee.Reducer.mean(),
      scale: 9000
    });

    return stats.map(function(f) {
      return f.set({
        'Year':     year,
        'District': f.get('NAME_3'),
        'VPD_kPa':  f.get('mean')
      });
    });
  })
).flatten();

var finalVPD = vpdTable.select(['Year', 'District', 'VPD_kPa']);

print('VPD Table (first 20):', finalVPD.limit(20));
print('Total rows:', finalVPD.size());

// Null check
var nullCheck = finalVPD.filter(ee.Filter.notNull(['VPD_kPa']));
print('Rows with valid VPD (should equal total rows):', nullCheck.size());

// Sanity check
// Expected range for Multan March-April: 0.5 – 2.5 kPa
// Minimum should now be exactly 0 or above — never negative
var vpdStats = finalVPD.aggregate_stats('VPD_kPa');
print('VPD statistics (min should be >= 0):', vpdStats);

// Batch 1: 2001–2010
// =====================================================

var districts = ee.FeatureCollection(

// Load ERA5-Land daily maximum temperature
var era5Tmax = ee.ImageCollection('ECMWF/ERA5_LAND/DAILY_AGGR')
  .select('temperature_2m_max');

// =====================================================
// CHANGE ONLY THIS LINE FOR EACH BATCH
// Batch 1: 2001–2010
// Batch 2: 2011–2020
// Batch 3: 2021–2025
// =====================================================
var years = ee.List.sequence(2021, 2025);

var tmaxTable = ee.FeatureCollection(
  years.map(function(year) {
    year = ee.Number(year);

    // March 1 to April 30 of same year
    var startDate = ee.Date.fromYMD(year, 3, 1);
    var endDate   = ee.Date.fromYMD(year, 4, 30);

    // Filter to March-April season
    var seasonCollection = era5Tmax
      .filterDate(startDate, endDate.advance(1, 'day'));

    // -------------------------------------------------
    // Variable 1: Mean of daily Tmax
    // Convert Kelvin to Celsius
    // -------------------------------------------------
    var tmaxImage = seasonCollection
      .mean()
      .subtract(273.15)
      .rename('Tmax_mean_C');

    // -------------------------------------------------
    // Variable 2: Heat Stress Days (Tmax > 34°C)
    // 34°C = 307.15 Kelvin
    // Each daily image returns 1 if > 34°C, else 0
    // Sum gives total count of heat stress days
    // Expected range for Multan March-April: 10–50 days
    // -------------------------------------------------
    var heatStressDays = seasonCollection
      .map(function(img) {
        return img
          .gt(307.15)        // 34 + 273.15 = 307.15 Kelvin
          .rename('HeatStress_days');
      })
      .sum()
      .rename('HeatStress_days');

    // -------------------------------------------------
    // Combine both variables into one image
    // -------------------------------------------------
    var combined = tmaxImage.addBands(heatStressDays);

    // Zonal statistics per district
    var stats = combined.reduceRegions({
      collection: districts,
      reducer: ee.Reducer.mean(),
      scale: 9000
    });

    return stats.map(function(f) {
      return f.set({
        'Year':             year,
        'District':         f.get('NAME_3'),
        'Tmax_mean_C':      f.get('Tmax_mean_C'),
        'HeatStress_days':  f.get('HeatStress_days')
      });
    });
  })
).flatten();

// =====================================================
// MINIMUM TEMPERATURE + COLD STRESS DAYS
// Source: ERA5-Land Daily Aggregated
// Window: November 1 (prev year) – February 28 (current year)
// Variable 1: Mean of daily Tmin (Celsius)
// Variable 2: Count of days where Tmin < 5°C
// Agronomic importance: Cold stress below 5°C
// damages germination and tillering in wheat
// Critical for Multan Division establishment phase
// Batch 1: 2001–2010
// =====================================================

var districts = ee.FeatureCollection(
  'projects/ee-ahsanahsan520/assets/multan_districts'
).filter(ee.Filter.inList('NAME_3', [
  'Multan', 'Lodhran', 'Khanewal', 'Vehari'
]));

// Load ERA5-Land daily minimum temperature
var era5Tmin = ee.ImageCollection('ECMWF/ERA5_LAND/DAILY_AGGR')
  .select('temperature_2m_min');

// =====================================================
// CHANGE ONLY THIS LINE FOR EACH BATCH
// Batch 1: 2001–2010
// Batch 2: 2011–2020
// Batch 3: 2021–2025
// =====================================================
var years = ee.List.sequence(2021, 2025);

var tminTable = ee.FeatureCollection(
  years.map(function(year) {
    year = ee.Number(year);

    // November of PREVIOUS year to February of current year
    // Wheat establishment and tillering phase
    var startDate = ee.Date.fromYMD(year.subtract(1), 11, 1);
    var endDate   = ee.Date.fromYMD(year, 2, 28);

    // Filter to Nov-Feb season
    var seasonCollection = era5Tmin
      .filterDate(startDate, endDate.advance(1, 'day'));

    // -------------------------------------------------
    // Variable 1: Mean of daily Tmin
    // Convert Kelvin to Celsius
    // -------------------------------------------------
    var tminImage = seasonCollection
      .mean()
      .subtract(273.15)
      .rename('Tmin_mean_C');

    // -------------------------------------------------
    // Variable 2: Cold Stress Days (Tmin < 5°C)
    // 5°C = 278.15 Kelvin
    // Each daily image returns 1 if < 5°C, else 0
    // Sum gives total count of cold stress days
    // Expected range for Multan Nov-Feb: 5–40 days
    // Below 5°C damages germination and tillering
    // Below 0°C causes frost damage to young plants
    // -------------------------------------------------
    var coldStressDays = seasonCollection
      .map(function(img) {
        return img
          .lt(278.15)        // 5 + 273.15 = 278.15 Kelvin
          .rename('ColdStress_days');
      })
      .sum()
      .rename('ColdStress_days');

    // -------------------------------------------------
    // Combine both variables into one image
    // -------------------------------------------------
    var combined = tminImage.addBands(coldStressDays);

    // Zonal statistics per district
    var stats = combined.reduceRegions({
      collection: districts,
      reducer: ee.Reducer.mean(),
      scale: 9000
    });

    return stats.map(function(f) {
      return f.set({
        'Year':            year,
        'District':        f.get('NAME_3'),
        'Tmin_mean_C':     f.get('Tmin_mean_C'),
        'ColdStress_days': f.get('ColdStress_days')
      });
    });
  })
).flatten();

var finalTMIN = tminTable.select([
  'Year', 'District', 'Tmin_mean_C', 'ColdStress_days'
]);

print('Tmin + Cold Stress Days Table (first 20):', 
      finalTMIN.limit(20));
print('Total rows:', finalTMIN.size());

// Null checks
var nullCheckTmin = finalTMIN.filter(
  ee.Filter.notNull(['Tmin_mean_C'])
);
print('Rows with valid Tmin (should equal total):', 
      nullCheckTmin.size());

var nullCheckCS = finalTMIN.filter(

);

      nullCheckCS.size());

print('Tmin statistics (expected 2-15 C):', tminStats);
